stateDiagram-v2
  [*] --> Importing

  state Importing {
    [*] --> Ingest
    Ingest --> Parse : file/doi/url received
    Parse --> Segment : sections/refs/figures extracted
    Segment --> Ready
    Ready --> [*]
  }

  Importing --> IntentSelection : parse ok
  Importing --> ImportError : parse fail

  ImportError --> Ingest : retry
  ImportError --> [*] : user cancels

  IntentSelection --> Skim : intent=skim
  IntentSelection --> DeepRead : intent=deep
  IntentSelection --> Review : intent=review
  IntentSelection --> Compare : intent=compare
  IntentSelection --> Dialogue : intent=dialogue

  %% Skim
  state Skim {
    [*] --> ScanTitleAbs
    ScanTitleAbs --> ScanFigures : quick figure sweep
    ScanFigures --> SkimOutput : generate SkimCard
    SkimOutput --> Decision
    Decision --> [*]
  }
  Skim --> IntentSelection : user switches intent

  %% Deep Read
  state DeepRead {
    [*] --> PlanRoute : pick must-read sections + key figs
    PlanRoute --> Walkthrough
    Walkthrough --> MethodReconstruct
    MethodReconstruct --> EvidenceAudit
    EvidenceAudit --> ReproPack
    ReproPack --> NotesFinalize : generate cards + checklists
    NotesFinalize --> [*]

    %% Time management
    Walkthrough --> Paused : user pause / timebox end
    Paused --> Walkthrough : resume
  }
  DeepRead --> IntentSelection : exit to other mode

  %% Review Mode
  state Review {
    [*] --> ClaimExtraction : list claims -> evidence pointers
    ClaimExtraction --> RigorCheck : baselines/ablations/stats/leakage
    RigorCheck --> ReviewOutput : reviewer report + questions
    ReviewOutput --> [*]
  }
  Review --> IntentSelection

  %% Compare Mode
  state Compare {
    [*] --> RetrievePeers : similar/contradict/alternate
    RetrievePeers --> Normalize : align terms/metrics/datasets
    Normalize --> MatrixBuild : comparison matrix + conflict map
    MatrixBuild --> [*]
  }
  Compare --> IntentSelection

  %% Dialogue Loop
  state Dialogue {
    [*] --> Chat
    Chat --> Chat : ask/answer/follow-up
    Chat --> [*] : user exits
  }
  Dialogue --> IntentSelection

  %% Module 2: Enhancement Engine as Substate (callable anywhere)
  state Enhance <<choice>>
  IntentSelection --> Enhance : user asks "解释/补背景/看公式/读图"
  Skim --> Enhance : unknown term / confusion
  DeepRead --> Enhance : equation/figure/detail missing
  Review --> Enhance : evidence ambiguity
  Compare --> Enhance : term alignment

  Enhance --> IntentSelection : return to caller state
